    ```javascript
    // Settings API
    async getSettings() {
      const response = await fetch('/settings');
      if (!response.ok) throw new Error('Failed to load settings');
      return response.json();
    },

    async getCategorySettings(category) {
      const response = await fetch(`/settings/${category}`);
      if (!response.ok) throw new Error(`Failed to load ${category} settings`);
      return response.json();
    },

    async updateSetting(category, key, value) {
      const response = await fetch('/settings', {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ category, key, value })
      });
      if (!response.ok) {
        const error = await response.json();
        throw new Error(error.detail || 'Failed to update setting');
      }
      return response.json();
    },

    async resetSettings() {
      const response = await fetch('/settings/reset', { method: 'POST' });
      if (!response.ok) throw new Error('Failed to reset settings');
      return response.json();
    }
    ```

    Add these methods to the API object, maintaining the existing pattern

    ```javascript
    import { API } from '../api/client.js';

    class SettingsComponent extends HTMLElement {
      constructor() {
        super();
        this.settings = null;
        this.originalSettings = null;
      }

      async connectedCallback() {
        await this.loadSettings();
        this.render();
        this.attachEventListeners();
      }

      async loadSettings() {
        try {
          this.settings = await API.getSettings();
          this.originalSettings = JSON.parse(JSON.stringify(this.settings));
        } catch (error) {
          console.error('Failed to load settings:', error);
          this.settings = this.getDefaultSettings();
        }
      }

      getDefaultSettings() {
        return {
          api: { engine_type: 'hybrid', timeout: 30, max_retries: 3 },
          cache: { enabled: true, ttl_seconds: 3600, max_entries: 1000 },
          rate_limiting: { default_tier: 'default', trial_requests: 20 },
          ui_preferences: { theme: 'dark', refresh_interval: 5, compact_mode: false }
        };
      }

      render() {
        this.innerHTML = `
          <div class="space-y-6">
            ${this.renderApiSettings()}
            ${this.renderCacheSettings()}
            ${this.renderRateLimitingSettings()}
            ${this.renderUISettings()}
            ${this.renderActions()}
          </div>
        `;
      }

      renderApiSettings() {
        const { api } = this.settings;
        return `
          <div class="p-6 border border-white/10 rounded-2xl bg-white/5">
            <h3 class="text-xl font-bold mb-4 flex items-center gap-2">
              <span class="material-symbols-outlined">api</span>
              API Configuration
            </h3>
            <div class="grid gap-4">
              <div>
                <label class="block text-sm text-white/60 mb-1">Engine Type</label>
                <select id="api-engine-type" class="w-full bg-white/10 border border-white/20 rounded-lg px-4 py-2">
                  <option value="hybrid" ${api.engine_type === 'hybrid' ? 'selected' : ''}>Hybrid (Vector + Graph)</option>
                  <option value="vector" ${api.engine_type === 'vector' ? 'selected' : ''}>Vector Only</option>
                </select>
              </div>
              <div>
                <label class="block text-sm text-white/60 mb-1">Timeout (seconds)</label>
                <input type="number" id="api-timeout" value="${api.timeout}" min="5" max="120"
                  class="w-full bg-white/10 border border-white/20 rounded-lg px-4 py-2">
              </div>
              <div>
                <label class="block text-sm text-white/60 mb-1">Max Retries</label>
                <input type="number" id="api-max-retries" value="${api.max_retries}" min="0" max="10"
                  class="w-full bg-white/10 border border-white/20 rounded-lg px-4 py-2">
              </div>
            </div>
          </div>
        `;
      }

      renderCacheSettings() {
        const { cache } = this.settings;
        return `
          <div class="p-6 border border-white/10 rounded-2xl bg-white/5">
            <h3 class="text-xl font-bold mb-4 flex items-center gap-2">
              <span class="material-symbols-outlined">storage</span>
              Cache Settings
            </h3>
            <div class="grid gap-4">
              <div class="flex items-center gap-3">
                <input type="checkbox" id="cache-enabled" ${cache.enabled ? 'checked' : ''}
                  class="w-5 h-5 accent-cyan-400">
                <label for="cache-enabled" class="text-white/80">Enable Caching</label>
              </div>
              <div>
                <label class="block text-sm text-white/60 mb-1">TTL (seconds)</label>
                <input type="number" id="cache-ttl" value="${cache.ttl_seconds}" min="60" max="86400"
                  class="w-full bg-white/10 border border-white/20 rounded-lg px-4 py-2">
              </div>
              <div>
                <label class="block text-sm text-white/60 mb-1">Max Entries</label>
                <input type="number" id="cache-max-entries" value="${cache.max_entries}" min="100" max="10000"
                  class="w-full bg-white/10 border border-white/20 rounded-lg px-4 py-2">
              </div>
            </div>
          </div>
        `;
      }

      renderRateLimitingSettings() {
        const { rate_limiting } = this.settings;
        return `
          <div class="p-6 border border-white/10 rounded-2xl bg-white/5">
            <h3 class="text-xl font-bold mb-4 flex items-center gap-2">
              <span class="material-symbols-outlined">speed</span>
              Rate Limiting
            </h3>
            <div class="grid gap-4">
              <div>
                <label class="block text-sm text-white/60 mb-1">Default Tier</label>
                <select id="rate-tier" class="w-full bg-white/10 border border-white/20 rounded-lg px-4 py-2">
                  <option value="trial" ${rate_limiting.default_tier === 'trial' ? 'selected' : ''}>Trial (20 req/hr)</option>
                  <option value="default" ${rate_limiting.default_tier === 'default' ? 'selected' : ''}>Default (100 req/hr)</option>
                  <option value="premium" ${rate_limiting.default_tier === 'premium' ? 'selected' : ''}>Premium (1000 req/hr)</option>
                </select>
              </div>
              <div>
                <label class="block text-sm text-white/60 mb-1">Trial Requests</label>
                <input type="number" id="rate-trial-requests" value="${rate_limiting.trial_requests}" min="5" max="100"
                  class="w-full bg-white/10 border border-white/20 rounded-lg px-4 py-2">
              </div>
            </div>
          </div>
        `;
      }

      renderUISettings() {
        const { ui_preferences } = this.settings;
        return `
          <div class="p-6 border border-white/10 rounded-2xl bg-white/5">
            <h3 class="text-xl font-bold mb-4 flex items-center gap-2">
              <span class="material-symbols-outlined">palette</span>
              UI Preferences
            </h3>
            <div class="grid gap-4">
              <div class="flex items-center gap-3">
                <input type="checkbox" id="ui-compact" ${ui_preferences.compact_mode ? 'checked' : ''}
                  class="w-5 h-5 accent-cyan-400">
                <label for="ui-compact" class="text-white/80">Compact Mode</label>
              </div>
              <div>
                <label class="block text-sm text-white/60 mb-1">Refresh Interval (seconds)</label>
                <input type="number" id="ui-refresh" value="${ui_preferences.refresh_interval}" min="1" max="60"
                  class="w-full bg-white/10 border border-white/20 rounded-lg px-4 py-2">
              </div>
            </div>
          </div>
        `;
      }

      renderActions() {
        return `
          <div class="flex gap-4 pt-4">
            <button id="save-settings" class="px-6 py-2 bg-cyan-500 hover:bg-cyan-400 text-black font-semibold rounded-lg transition-colors">
              Save Changes
            </button>
            <button id="reset-settings" class="px-6 py-2 bg-white/10 hover:bg-white/20 text-white font-semibold rounded-lg transition-colors">
              Reset to Defaults
            </button>
          </div>
          <div id="settings-message" class="hidden mt-4 p-3 rounded-lg"></div>
        `;
      }

      attachEventListeners() {
        const saveBtn = this.querySelector('#save-settings');
        const resetBtn = this.querySelector('#reset-settings');

        if (saveBtn) {
          saveBtn.addEventListener('click', () => this.saveSettings());
        }
        if (resetBtn) {
          resetBtn.addEventListener('click', () => this.resetSettings());
        }
      }

      showMessage(text, isError = false) {
        const msg = this.querySelector('#settings-message');
        if (msg) {
          msg.textContent = text;
          msg.className = `mt-4 p-3 rounded-lg ${isError ? 'bg-red-500/20 text-red-400' : 'bg-green-500/20 text-green-400'}`;
          msg.classList.remove('hidden');
          setTimeout(() => msg.classList.add('hidden'), 3000);
        }
      }

      async saveSettings() {
        try {
          // Collect all values
          const updates = [
            { category: 'api', key: 'engine_type', value: this.querySelector('#api-engine-type').value },
            { category: 'api', key: 'timeout', value: parseInt(this.querySelector('#api-timeout').value) },
            { category: 'api', key: 'max_retries', value: parseInt(this.querySelector('#api-max-retries').value) },
            { category: 'cache', key: 'enabled', value: this.querySelector('#cache-enabled').checked },
            { category: 'cache', key: 'ttl_seconds', value: parseInt(this.querySelector('#cache-ttl').value) },
            { category: 'cache', key: 'max_entries', value: parseInt(this.querySelector('#cache-max-entries').value) },
            { category: 'rate_limiting', key: 'default_tier', value: this.querySelector('#rate-tier').value },
            { category: 'rate_limiting', key: 'trial_requests', value: parseInt(this.querySelector('#rate-trial-requests').value) },
            { category: 'ui_preferences', key: 'compact_mode', value: this.querySelector('#ui-compact').checked },
            { category: 'ui_preferences', key: 'refresh_interval', value: parseInt(this.querySelector('#ui-refresh').value) }
          ];

          for (const update of updates) {
            await API.updateSetting(update.category, update.key, update.value);
          }

          this.showMessage('Settings saved successfully!');
          this.originalSettings = JSON.parse(JSON.stringify(this.settings));
        } catch (error) {
          console.error('Failed to save settings:', error);
          this.showMessage(`Error: ${error.message}`, true);
        }
      }

      async resetSettings() {
        if (!confirm('Are you sure you want to reset all settings to defaults?')) {
          return;
        }

        try {
          await API.resetSettings();
          await this.loadSettings();
          this.render();
          this.attachEventListeners();
          this.showMessage('Settings reset to defaults');
        } catch (error) {
          console.error('Failed to reset settings:', error);
          this.showMessage(`Error: ${error.message}`, true);
        }
      }
    }

    customElements.define('settings-component', SettingsComponent);
    ```
